# HW5 — команды Spin/Promela
# Каждый LTL-предикат проверяется отдельным запуском pan.

# Исправная модель
cd src/main/java/HW5/queue_right
spin -a queue.pml
gcc -O2 -o pan pan.c

mkdir -p results
rm -f queue.pml.trail
./pan -a -f -N S1 > results/S1.txt 2>&1

rm -f queue.pml.trail
./pan -a -f -N S2 > results/S2.txt 2>&1

rm -f queue.pml.trail
./pan -a -f -N S3 > results/S3.txt 2>&1

rm -f queue.pml.trail
./pan -a -f -N L1 > results/L1.txt 2>&1

rm -f queue.pml.trail
./pan -a -f -N L2 > results/L2.txt 2>&1


cd ../queue_buggy
spin -a queue.pml
gcc -O2 -o pan pan.c
mkdir -p results traces

rm -f queue.pml.trail
./pan -a -f -N S1 > results/S1.txt 2>&1
test -f queue.pml.trail && cp queue.pml.trail traces/S1.trail
